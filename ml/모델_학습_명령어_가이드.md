## 12) Two-stage 튜닝

형식별 일괄
>
Set-Location "C:\Users\oyh57\SODAsoo1\Devs\UTAU_Auto_OTO_v3\Auto_OTO"
$env:PYTHONUTF8="1"
$env:PYTHONIOENCODING="utf-8"
>

function Run-TwoStageFormat {
  param(
    [string]$Lang,
    [string]$Fmt,
    [int]$Stage1Trials = 12,
    [int]$Stage2Trials = 8
  )

  $cfg = ".\scripts\coupled_two_stage_${Lang}_${Fmt}.yaml"

  @"
import copy, yaml
src = r".\scripts\coupled_two_stage_tune.sample.yaml"
cfg = yaml.safe_load(open(src, "r", encoding="utf-8")) or {}
cfg["run_tag"] = "two_stage_${Lang}_${Fmt}"
tm = cfg.setdefault("train_matrix", {})
args = tm.setdefault("args", {})
args["dataset-root"] = "dataset_staged"
args["languages"] = "${Lang}"
args["formats"] = "${Lang}=${Fmt}"
yaml.safe_dump(cfg, open(r"${cfg}", "w", encoding="utf-8"), sort_keys=False, allow_unicode=False)
print(r"${cfg}")
"@ | python -X utf8 -

  python -X utf8 .\scripts\run_coupled_two_stage_tune.py `
    --config $cfg `
    --stage1-max-trials $Stage1Trials `
    --stage2-max-trials $Stage2Trials
}
>
실행
# Korean
Run-TwoStageFormat korean cv
Run-TwoStageFormat korean cvc
Run-TwoStageFormat korean cvvc
Run-TwoStageFormat korean vcv

# Japanese
Run-TwoStageFormat japanese cv
Run-TwoStageFormat japanese cvvc
Run-TwoStageFormat japanese vcv
>

